def compute_network_interaction_matrix(agents: List[DimensionalState]) -> np.ndarray:
    """
    Compute cognitive coupling strengths based on complementarity and resonance
    Functional: f: S^n → [0,1]^(n×n)
    """
    n = len(agents)
    I = np.zeros((n, n))

    for i in range(n):
        for j in range(i+1, n):
            # Complementarity measure (how well they complete each other)
            complement = np.sum(agents[i].vec * (1 - agents[j].vec))

            # Resonance measure (frequency alignment)
            freq_align = 1 - abs(agents[i].freq - agents[j].freq) / max(agents[i].freq, agents[j].freq)

            # Order compatibility (adjacent orders interact more strongly)
            order_compat = 1 / (1 + abs(agents[i].order - agents[j].order))

            # Combined interaction strength
            strength = 0.4 * complement + 0.3 * freq_align + 0.3 * order_compat
            I[i,j] = I[j,i] = np.clip(strength, 0, 1)

    return I

def compute_collective_coherence(cycle_result: Dict) -> float:
    """
    Measure how well individual cognitive states align into collective patterns
    Metric: Φ = (1/n²) Σᵢⱼ ⟨ψᵢ|ψⱼ⟩ for all state pairs in cycle
    """
    states = cycle_result['states']
    if len(states) < 2:
        return 0.0

    coherence_sum = 0.0
    count = 0

    for i in range(len(states)):
        for j in range(i+1, len(states)):
            coherence_sum += abs(np.dot(states[i].vec, states[j].vec))
            count += 1

    return coherence_sum / count if count > 0 else 0.0

def compute_cognitive_complexity(cycle_result: Dict) -> float:
    """
    Measure the cognitive complexity of the collective behavior
    Metric: C = H(UC) + λ * H(Modes) where H is Shannon entropy
    """
    uc_values = cycle_result['uc_values']
    modes = cycle_result['triggered_modes']

    # UC entropy
    uc_mean = np.mean(uc_values, axis=0)
    uc_mean = uc_mean / np.sum(uc_mean)  # Normalize to probability distribution
    uc_entropy = -np.sum(np.where(uc_mean > 0, uc_mean * np.log(uc_mean), 0))

    # Mode entropy
    all_modes = [mode for mode_list in modes for mode in mode_list]
    if all_modes:
        mode_counts = {}
        for mode in all_modes:
            mode_counts[mode] = mode_counts.get(mode, 0) + 1

        mode_probs = np.array(list(mode_counts.values())) / len(all_modes)
        mode_entropy = -np.sum(mode_probs * np.log(mode_probs))
    else:
        mode_entropy = 0.0

    return 0.7 * uc_entropy + 0.3 * mode_entropy

def compute_modal_preference_profile(evolution_results: List[Dict]) -> Dict:
    """
    Extract statistical patterns in modal dynamic preferences
    """
    all_preferences = {}

    for result in evolution_results:
        for mode, count in result['dominant_modes']:
            if mode not in all_preferences:

    profile = {}

            'mean_activation': np.mean(counts),
            'activation_variability': np.std(counts),

        }

    return profile
